      function formatValue(value) {
        return value.toFixed(3);
      }

      function renderDataChips(item) {
        dataNode.innerHTML = item.labels.map(function (label, index) {
          return '<span class="data-chip">' + label + '<strong>μ ' + formatValue(item.values[index]) + '</strong></span>';
        }).join("");
      }

      function renderChart(item) {
        var width = 760;
        var height = 394;
        var left = 82;
        var right = 54;
        var top = 70;
        var bottom = 68;
        var plotW = width - left - right;
        var plotH = height - top - bottom;
        var min = Math.min.apply(null, item.values);
        var max = Math.max.apply(null, item.values);
        var pad = Math.max((max - min) * 0.22, 0.004);
        min = Math.max(0, min - pad);
        max = max + pad;

        function xAt(index) {
          if (item.values.length === 1) return left + plotW / 2;
          return left + (plotW * index) / (item.values.length - 1);
        }

        function yAt(value) {
          return top + (max - value) / (max - min) * plotH;
        }

        var grid = [0, 1, 2, 3].map(function (i) {
          var y = top + plotH * i / 3;
          return '<line class="grid" x1="' + left + '" y1="' + y + '" x2="' + (left + plotW) + '" y2="' + y + '"></line>';
        }).join("");

        var xLabels = item.labels.map(function (label, index) {
          return '<text x="' + xAt(index) + '" y="' + (height - 26) + '" text-anchor="middle">' + label + '</text>';
        }).join("");

        var yLabels = [max, (max + min) / 2, min].map(function (value, index) {
          var y = index === 0 ? top + 8 : index === 1 ? top + plotH / 2 + 6 : top + plotH + 4;
          return '<text x="' + (left - 18) + '" y="' + y + '" text-anchor="end">' + formatValue(value) + '</text>';
        }).join("");

        var body = "";
        if (item.type === "bar") {
          var barW = 70;
          body = item.values.map(function (value, index) {
            var x = xAt(index) - barW / 2;
            var y = yAt(value);
            var h = top + plotH - y;
            var cls = index === item.values.length - 1 ? "bar bar-accent" : "bar";
            return '<rect class="' + cls + '" x="' + x + '" y="' + y + '" width="' + barW + '" height="' + h + '" rx="10"></rect>';
          }).join("");
          body += '<path class="trend" d="' + item.values.map(function (value, index) {
            return (index === 0 ? "M" : "L") + xAt(index) + " " + yAt(value);
          }).join(" ") + '"></path>';
        } else {
          var path = item.values.map(function (value, index) {
            return (index === 0 ? "M" : "L") + xAt(index) + " " + yAt(value);
          }).join(" ");
          var area = path + " L" + xAt(item.values.length - 1) + " " + (top + plotH) + " L" + xAt(0) + " " + (top + plotH) + " Z";
          body = '<path class="area" d="' + area + '"></path><path class="trend" d="' + path + '"></path>';
        }

        body += item.values.map(function (value, index) {
          var cls = index === item.values.length - 1 ? "dot dot-accent" : "dot";
          return '<circle class="' + cls + '" cx="' + xAt(index) + '" cy="' + yAt(value) + '" r="8"></circle>';
        }).join("");

        chartNode.innerHTML =
          '<svg viewBox="0 0 ' + width + ' ' + height + '" role="img" aria-label="' + item.chartTitle + '">' +
          '<text class="chart-title" x="38" y="38">' + item.chartTitle + '</text>' +
          grid +
          '<line class="axis" x1="' + left + '" y1="' + (top + plotH) + '" x2="' + (left + plotW) + '" y2="' + (top + plotH) + '"></line>' +

          yLabels +
          xLabels +
          body +
          '</svg>';
      }

      function openAnalysis(key) {
        var item = factorData[key] || factorData.temperature;
